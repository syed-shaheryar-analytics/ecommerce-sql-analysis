 =========================================================
 E-COMMERCE SQL ANALYSIS PROJECT
 Dataset: users, orders, order_items, products
 Objective: Calculate revenue metrics and business insights
=========================================================


 1. TOTAL REVENUE
Calculates total revenue from all order items


SELECT SUM(quantity * item_price) AS total_revenue
FROM order_items;


 2. MONTHLY REVENUE
Calculates total revenue grouped by month


SELECT 
    strftime('%Y-%m', o.order_date) AS month,
    SUM(oi.quantity * oi.item_price) AS monthly_revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY month
ORDER BY month;


3. TOP 5 PRODUCTS BY REVENUE
Identifies highest revenue generating products


SELECT 
    p.product_name,
    SUM(oi.quantity * oi.item_price) AS revenue
FROM order_items oi
JOIN products p ON oi.product_id = p.product_id
GROUP BY p.product_name
ORDER BY revenue DESC
LIMIT 5;


4. REVENUE BY CITY
Shows which cities generate highest revenue


SELECT 
    u.city,
    SUM(oi.quantity * oi.item_price) AS revenue
FROM orders o
JOIN users u ON o.user_id = u.user_id
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY u.city
ORDER BY revenue DESC;


5. AVERAGE ORDER VALUE
Calculates average revenue per order using subquery


SELECT AVG(order_total) AS avg_order_value
FROM (
    SELECT 
        o.order_id,
        SUM(oi.quantity * oi.item_price) AS order_total
    FROM orders o
    JOIN order_items oi ON o.order_id = oi.order_id
    GROUP BY o.order_id
);
